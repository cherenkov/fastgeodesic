使い方

# import の geopy を fastgeodesic に変えるだけ
# from geopy.distance import geodesic
from fastgeodesic.distance import geodesic

geodesic((35.36, 138.73), (35.68, 139.77)).km
# import を差し替える場合
from fastgeodesic.geodesic import Geodesic
Geodesic.WGS84.Inverse(35.36, 138.73, 35.68, 139.77)["s12"]
